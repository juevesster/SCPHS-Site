  <script>
  (() => {
    const slider = document.getElementById('home-slider');
    if (!slider) return;

    const slides = Array.from(slider.querySelectorAll('.slide'));
    const dotsHost = slider.querySelector('.dots');
    const prevBtn = slider.querySelector('.prev');
    const nextBtn = slider.querySelector('.next');
    const interval = parseInt(slider.dataset.interval, 10) || 5000;

    // Build dots dynamically based on slides
    const dots = slides.map((_, i) => {
      const b = document.createElement('button');
      b.type = 'button';
      b.setAttribute('aria-label', `Go to slide ${i + 1}`);
      b.addEventListener('click', () => go(i, true));
      dotsHost.appendChild(b);
      return b;
    });

    let index = 0;
    let timer;

    function setActive(i){
      slides.forEach((s, j) => s.classList.toggle('is-active', j === i));
      dots.forEach((d, j) => d.setAttribute('aria-current', String(j === i)));
    }

    function next(){ go((index + 1) % slides.length); }
    function prev(){ go((index - 1 + slides.length) % slides.length); }

    function go(i, user=false){
      index = i;
      setActive(index);
      if (user) restart();
    }

    function start(){
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
      timer = setInterval(next, interval);
    }
    function stop(){ clearInterval(timer); }
    function restart(){ stop(); start(); }

    nextBtn.addEventListener('click', () => go((index + 1) % slides.length, true));
    prevBtn.addEventListener('click', () => go((index - 1 + slides.length) % slides.length, true));
    slider.addEventListener('mouseenter', stop);
    slider.addEventListener('mouseleave', start);
    slider.addEventListener('keydown', (e) => {
      if (e.key === 'ArrowRight') { e.preventDefault(); nextBtn.click(); }
      if (e.key === 'ArrowLeft')  { e.preventDefault(); prevBtn.click(); }
    });
    slider.tabIndex = 0; // focusable

    // Init
    setActive(0);
    start();
  })();
  </script>
