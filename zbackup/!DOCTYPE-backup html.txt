  <!-- Issuances feed from Google Sheet (Opensheet) -->
  <script>
  // --- CONFIG ---
  const SHEET_JSON = 'https://opensheet.elk.sh/1OKi8_qt3HOShs1JLWsP1UYAjlSt6FjjpoUilISLwQOA/Issuances';

  // --- HELPERS ---
  const esc = (s='') => String(s)
    .replaceAll('&','&amp;').replaceAll('<','&lt;')
    .replaceAll('>','&gt;').replaceAll('"','&quot;').replaceAll("'","&#39;");

  const fmtDate = (iso='') => {
    if (!iso) return '';
    const [y,m,d] = iso.split('-');
    const dt = new Date(+y, (+m||1)-1, +d||1);
    if (isNaN(dt)) return iso;
    return dt.toLocaleDateString('en-PH', { month:'short', day:'numeric', year:'numeric' });
  };

  // --- STATE ---
  const LIST     = document.getElementById('issuances-list');
  const FILTERS  = document.querySelector('.filter-bar');
  const LOADMORE = document.getElementById('load-more');

  let ALL_ITEMS    = [];
  let currentLevel = 'division';   // default
  let pageSize     = 20;
  let shownCount   = 0;
  const AUTO_REFRESH_MS = 10 * 60 * 1000; // 10 minutes

  function render(items) {
    if (!items.length) {
      LIST.innerHTML = '<div class="meta">No issuances found.</div>';
      return;
    }
    LIST.innerHTML = items.map(row => {
      const title = esc(row.title || 'Untitled');
      const url   = esc(row.url   || '#');
      const date  = fmtDate(row.date || '');
      const src   = esc(row.source || '');
      const series= esc((row.series || '').toUpperCase());
      const level = row.level ? ' • ' + esc(String(row.level).toUpperCase()) : '';
      const type  = row.type  ? ' • ' + esc(row.type) : '';
      return `
        <article class="post">
          <div class="meta">${date} • ${src}</div>
          <h4><a href="${url}" target="_blank" rel="noopener noreferrer">${title}</a></h4>
          <div class="meta">${series}${level}${type}</div>
        </article>
      `;
    }).join('');
  }

  function getFiltered() {
    let rows = ALL_ITEMS.slice();
    if (currentLevel) {
      rows = rows.filter(r => String((r.level||'')).toLowerCase() === currentLevel);
    }
    rows.sort((a,b) =>
      String(b.date||'').localeCompare(String(a.date||'')) ||
      String(b.title||'').localeCompare(String(a.title||''))
    );
    return rows;
  }

  function showMore() {
    const rows = getFiltered();
    const slice = rows.slice(0, shownCount + pageSize);
    shownCount = slice.length;
    render(slice);
    if (LOADMORE) {
      LOADMORE.disabled = (shownCount >= rows.length);
      LOADMORE.textContent = LOADMORE.disabled ? 'All items shown' : 'Load more';
    }
  }

  // --- LOAD FROM SHEET JSON ---
  async function loadSheet({silent=false} = {}) {
    try {
      if (!silent) LIST.innerHTML = '<div class="meta">Loading latest memoranda &amp; orders…</div>';
      const res = await fetch(SHEET_JSON, { cache: 'no-store' });
      if (!res.ok) throw new Error('HTTP ' + res.status);
      const data = await res.json();

      if (!Array.isArray(data)) throw new Error('Sheet JSON not an array');

      // Normalize keys to lower-case (tolerates header casing)
      ALL_ITEMS = data.map(r => {
        const o = {};
        Object.keys(r).forEach(k => o[k.toLowerCase()] = r[k]);
        return o;
      });

      // 🔔 Notify right sidebar (Categories & Archives) that data is available
      document.dispatchEvent(new CustomEvent('feed:loaded', { detail: { items: ALL_ITEMS } }));

      shownCount = 0;
      showMore();
    } catch (e) {
      console.error(e);
      LIST.innerHTML = `
        <div class="meta" style="color:#b91c1c;">Failed to load issuances.</div>
        <div class="meta">
          If this persists, open sources:
          <a href="https://www.deped.gov.ph/category/issuances/deped-orders/" target="_blank" rel="noopener">DepEd Orders</a> •
          <a href="https://www.deped.gov.ph/deped-memorandum/" target="_blank" rel="noopener">DepEd Memoranda</a> •
          <a href="https://sdonuevavizcaya.com/category/division-memoranda/" target="_blank" rel="noopener">SDO NV Memoranda</a>
        </div>
      `;
    }
  }

  // --- EVENTS ---
  FILTERS?.addEventListener('click', e => {
    const btn = e.target.closest('.pill[data-level]');
    if (!btn) return;
    FILTERS.querySelectorAll('.pill').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
    currentLevel = btn.dataset.level; // '' | 'national' | 'division'
    shownCount = 0;
    showMore();
  });

  LOADMORE?.addEventListener('click', () => showMore());

  // --- BOOT ---
  loadSheet();
  setInterval(() => loadSheet({silent:true}), AUTO_REFRESH_MS);
  </script>

  <!-- Sidebar builder: Category counts + Archives -->
  <script>
  (function () {
    const catHost  = document.querySelector('#categories .list');
    const archHost = document.querySelector('#archives .list');
    if (!catHost && !archHost) return;

    function matcherFromLink(a) {
      const u = new URL(a.getAttribute('href'), location.href);
      const p = u.searchParams;
      const cat   = (p.get('cat') || '').toLowerCase();
      const level = (p.get('level') || '').toLowerCase();
      const type  = (p.get('type')  || '').toLowerCase();

      let resolvedLevel = level;
      let resolvedType  = type;

      if (!resolvedLevel) {
        if (cat === 'national') resolvedLevel = 'national';
        if (cat === 'division') resolvedLevel = 'division';
      }
      if (!resolvedType) {
        if (cat === 'orders' || cat === 'order') resolvedType = 'order';
        if (cat === 'memoranda' || cat === 'memorandum') resolvedType = 'memorandum';
        if (cat === 'division-memos') resolvedType = 'memorandum';
      }

      return function (r) {
        const lv = String(r.level || '').toLowerCase();
        const tp = String(r.type  || '').toLowerCase();
        let ok = true;
        if (resolvedLevel) ok = ok && (lv === resolvedLevel);
        if (cat === 'division-memos') ok = ok && (lv === 'division');
        if (resolvedType) ok = ok && (tp === resolvedType);
        return ok;
      };
    }

    function updateCategoryCounts(items) {
      if (!catHost) return;
      catHost.querySelectorAll('a').forEach(a => {
        const match = matcherFromLink(a);
        const count = items.filter(match).length;
        const countEl = a.querySelector('.count');
        if (countEl) {
          countEl.textContent = count ? `(${count})` : '';
          countEl.style.color = 'var(--muted)';
        }
      });
    }

    function buildArchives(items, maxMonths = 12) {
      if (!archHost) return;

      const counts = new Map();
      items.forEach(r => {
        const iso = String(r.date || '').trim();
        if (!/^\d{4}-\d{2}/.test(iso)) return;
        const ym = iso.slice(0, 7); // YYYY-MM
        counts.set(ym, (counts.get(ym) || 0) + 1);
      });

      const months = Array.from(counts.keys())
        .sort((a,b) => b.localeCompare(a))
        .slice(0, maxMonths);

      archHost.innerHTML = months.length
        ? months.map(ym => {
            const [y,m] = ym.split('-');
            const dt = new Date(+y, +m - 1, 1);
            const label = dt.toLocaleDateString('en-PH', { month:'long', year:'numeric' });
            const href  = `issuances.html?year=${y}&month=${m}`;
            const n     = counts.get(ym);
            return `<li><a href="${href}">${label} <span class="meta" style="color:var(--muted)">(${n})</span></a></li>`;
          }).join('')
        : '<li><span class="meta">No archives yet</span></li>';
    }

    // Rebuild when the feed is loaded/refreshed
    document.addEventListener('feed:loaded', (e) => {
      const items = e.detail?.items || [];
      updateCategoryCounts(items);
      buildArchives(items);
    });

    // If data already exists (first paint/cache), run once
    if (window.ALL_ITEMS && Array.isArray(window.ALL_ITEMS) && ALL_ITEMS.length) {
      updateCategoryCounts(ALL_ITEMS);
      buildArchives(ALL_ITEMS);
    }
  })();
  </script>

  <script>
  (() => {
    const slider = document.getElementById('home-slider');
    if (!slider) return;
    const slides = Array.from(slider.querySelectorAll('.slide'));
    const dotsHost = slider.querySelector('.dots');
    const prevBtn = slider.querySelector('.prev');
    const nextBtn = slider.querySelector('.next');
    const interval = parseInt(slider.dataset.interval, 10) || 5000;

    const dots = slides.map((_, i) => {
      const b = document.createElement('button');
      b.type = 'button';
      b.setAttribute('aria-label', 'Go to slide ' + (i+1));
      b.addEventListener('click', () => go(i, true));
      dotsHost.appendChild(b);
      return b;
    });

    let index = 0, timer;
    const setActive = i => {
      slides.forEach((s, j) => s.classList.toggle('is-active', j === i));
      dots.forEach((d, j) => d.setAttribute('aria-current', String(j === i)));
    };
    const next = () => go((index + 1) % slides.length);
    const prev = () => go((index - 1 + slides.length) % slides.length);
    const go = (i, user=false) => { index = i; setActive(i); if (user) { clearInterval(timer); start(); } };
    const start = () => {
      if (!window.matchMedia('(prefers-reduced-motion: reduce)').matches) timer = setInterval(next, interval);
    };

    prevBtn.addEventListener('click', () => go((index - 1 + slides.length) % slides.length, true));
    nextBtn.addEventListener('click', () => go((index + 1) % slides.length, true));
    slider.addEventListener('mouseenter', () => clearInterval(timer));
    slider.addEventListener('mouseleave', start);
    slider.tabIndex = 0;

    setActive(0); start();
  })();
  </script>

  <!-- Dropdown Nav JS (The School) + footer year -->
  <script>
  (function () {
    const dropdowns = document.querySelectorAll('.menu .has-dropdown');

    dropdowns.forEach(dd => {
      const btn = dd.querySelector('.menu-toggle');
      const panel = dd.querySelector('.dropdown');

      function setOpen(open) {
        dd.classList.toggle('open', open);
        btn.setAttribute('aria-expanded', String(open));
      }

      btn.addEventListener('click', (e) => {
        e.stopPropagation();
        setOpen(!dd.classList.contains('open'));
      });

      // Close on outside click
      document.addEventListener('click', (e) => {
        if (!dd.contains(e.target)) setOpen(false);
      });

      // Close via Esc
      dd.addEventListener('keydown', (e) => {
        if (e.key === 'Escape') {
          setOpen(false);
          btn.focus();
        }
      });

      // Close when a submenu link is clicked (good for mobile)
      panel.addEventListener('click', (e) => {
        if (e.target.closest('a')) setOpen(false);
      });

      // Close when focus leaves the dropdown completely
      panel.addEventListener('focusout', (e) => {
        if (!dd.contains(e.relatedTarget)) setOpen(false);
      });
    });

    // Set year in GOVPH footer bar
    const y = document.getElementById('yearNow');

<script>
/**
 * Tiny click-to-open helper for nav dropdowns.
 * Works with your HTML structure:
 * <li class="has-dropdown">
 *   <button class="menu-toggle" aria-expanded="false">…</button>
 *   <ul class="dropdown …">…</ul>
 * </li>
 */
(function () {
  const ROOT = document;
  const OPEN_CLASS = 'open';
  const ITEM_SELECTOR = '.menu .has-dropdown';
  const TOGGLE_SELECTOR = '.menu .has-dropdown > .menu-toggle';

  // Toggle on click/tap
  ROOT.querySelectorAll(TOGGLE_SELECTOR).forEach((btn) => {
    btn.addEventListener('click', (e) => {
      const li = btn.closest(ITEM_SELECTOR);
      const isOpen = li.classList.contains(OPEN_CLASS);

      // Close others first (keeps only one open)
      ROOT.querySelectorAll(`${ITEM_SELECTOR}.${OPEN_CLASS}`).forEach((openItem) => {
        if (openItem !== li) {
          openItem.classList.remove(OPEN_CLASS);
          const toggle = openItem.querySelector('.menu-toggle');
          if (toggle) toggle.setAttribute('aria-expanded', 'false');
        }
      });

      // Toggle current
      li.classList.toggle(OPEN_CLASS, !isOpen);
      btn.setAttribute('aria-expanded', String(!isOpen));
    });
  });

  // Close when clicking outside
  ROOT.addEventListener('click', (e) => {
    if (!e.target.closest(ITEM_SELECTOR)) {
      ROOT.querySelectorAll(`${ITEM_SELECTOR}.${OPEN_CLASS}`).forEach((openItem) => {
        openItem.classList.remove(OPEN_CLASS);
        const toggle = openItem.querySelector('.menu-toggle');
        if (toggle) toggle.setAttribute('aria-expanded', 'false');
      });
    }
  });

  // Close on Escape
  ROOT.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') {
      ROOT.querySelectorAll(`${ITEM_SELECTOR}.${OPEN_CLASS}`).forEach((openItem) => {
        openItem.classList.remove(OPEN_CLASS);
        const toggle = openItem.querySelector('.menu-toggle');
        if (toggle) toggle.setAttribute('aria-expanded', 'false');
      });
    }
  });
})();
</script>
